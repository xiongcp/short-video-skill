# 上线审查与发布计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

- 日期：2026-11-26
- 状态：待 owner 审阅第 2 节的决定后开工
- 分支：`lite/launch`（从 `main` 的 e2a11fe 切出）

**Goal:** 让 topic-to-video 达到可以对外上线的状态：用户说一句话或一个主题，环境具备时两轮对话内拿到本机渲染、与主题相关的 1080p60 解说成片；环境不具备时稳定退回文本包；README、官网、安装部署与使用文档和实际行为一致；完成三家 Agent 实测后发布 v1.0.0。

**Architecture:** 渲染核心已经可用，不重写，只补"一句话到成片"链路上的缺口：`SKILL.md` 把 L1 改成环境具备时的默认交付，新增 `references/render.md` 写清 L1 出片步骤；`render.mjs` 改为输出结构化错误并在出片前自检；`doctor.mjs` 补上配音与中文字体判断；按计划书 §4.5 给未经确认的成片加"未审核初稿"角标。然后修一个可见的画面缺陷、补通用科普物件，用两个新主题做本机验收，最后重写 README、官网、安装部署与使用文档，加 CI，跑三家实测，发版。

**Tech Stack:** 开发用 Node ≥ 22（ESM、`node --test`），skill 运行时 Node ≥ 20；`@resvg/resvg-js`（唯一运行时依赖）；FFmpeg/FFprobe；edge-tts（经 `uvx`）；GitHub Actions、GitHub Pages。

**Spec:** [轻量化 Skill 计划书](轻量化Skill计划书.md) §4.5、§7.3、P4；[超越路线图](超越路线图.md) M6–M7 与 §7；[个性化路线图](个性化路线图.md) §6.6；本文第 1、2 节。

export const ERROR_CODES = ['PROJECT_NOT_FOUND', 'PROJECT_INVALID', 'VOICE_UNDECIDED', 'TTS_FAILED', 'AUDIO_MISSING', 'SAMPLE_RANGE_INVALID', 'RENDER_FAILED'];

export function errorReport(err, doctor, project) {
  const code = ERROR_CODES.includes(err?.code) ? err.code : 'RENDER_FAILED';
  const report = { ok: false, code, message: String(err?.message ?? err).split('\n')[0] };
  if (err?.errors) report.errors = err.errors.slice(0, 10);
  if (code === 'VOICE_UNDECIDED') {
    const voice = { ...doctor.recommended_voice };
    if (/edge-tts/.test(voice.command ?? '') && project?.voice?.preset) voice.preset = project.voice.preset;
    report.fix = { voice };
  }
  return report;
}

function readProject(dir) {
  const file = join(dir, 'project.json');
  if (!existsSync(file)) throw Object.assign(new Error(`${dir} 里没有 project.json`), { code: 'PROJECT_NOT_FOUND' });
  const project = JSON.parse(readFileSync(file, 'utf8'));
  const errors = checkProject(project);
  if (errors.length) throw Object.assign(new Error(`project.json 有 ${errors.length} 个错误，先运行 check.mjs 改好`), { code: 'PROJECT_INVALID', errors });
  return project;
}

const SKILL_DIRS = { claude: '.claude/skills', agy: '.agents/skills', codex: '.agents/skills' };
export function installSkill(work, agent, level) {
  const dest = join(work, SKILL_DIRS[agent], 'topic-to-video');
  cpSync(join(ROOT, 'skills/topic-to-video'), dest, { recursive: true, filter: (p) => !p.split(/[\\/]/).includes('node_modules') });

export function inspectRender(dir) {
  const video = join(dir, 'out/final.mp4');
  const qaFile = join(dir, 'out/qa.json');
  let final = null;
  if (existsSync(video)) {
    const probe = JSON.parse(execFileSync('ffprobe', ['-v', 'error', '-select_streams', 'v:0', '-show_entries', 'stream=width,height,r_frame_rate:format=duration', '-of', 'json', video], { encoding: 'utf8' }));
    const [num, den] = probe.streams[0].r_frame_rate.split('/').map(Number);
    final = { exists: true, width: probe.streams[0].width, height: probe.streams[0].height, fps: num / den, duration: Number(probe.format.duration) };
  }
  const qa = existsSync(qaFile) ? (({ ok, issues, draft }) => ({ ok, issues: issues?.length ?? 0, draft: Boolean(draft) }))(JSON.parse(readFileSync(qaFile, 'utf8'))) : null;
  return { final, qa };
}

// 复制工作区时：大文件不进 results/
const HEAVY = new Set(['out', 'cache', 'audio']);
const keep = (p) => {
  const parts = relative(work, p).split(/[\\/]/);
  if (parts.some((part) => SKIP.has(part))) return false;
  const i = parts.findIndex((part) => HEAVY.has(part));
  return i < 0 || (parts[i] === 'out' && (parts.length === i + 1 || ['qa.json', 'qa.md'].includes(parts[i + 1])));
};

export function chooseLinkLabelT({ p1, m, p2 }, { w, h }, host) {
